    <script>
        $(document).ready(function() {
            fetchFacultyList();

            $('#addFacultyForm').on('submit', function(e) {
                e.preventDefault();
                ajaxRequest('../database/Faculty/add_faculty.php', 'POST', $(this).serialize(), function(response) {
                    $('#addFacultyForm')[0].reset();
                    fetchFacultyList();
                    alert(response);
                    $('#basicModal').modal('hide'); // Close the modal
                });
            });

            $(document).on('click', '.edit-btn', function() {
                var facultyId = $(this).data('id');
                console.log("Editing faculty ID: ", facultyId);
                ajaxRequest('../database/Faculty/get_faculty.php', 'GET', { fid: facultyId }, function(response) {
                    var facultyData = JSON.parse(response);
                    $('#editFacultyId').val(facultyData.fid);
                    $('#editFacultyName').val(facultyData.faculty_name);
                    $('#editModal').modal('show');
                });
            });

            $(document).on('click', '.delete-btn', function() {
                var facultyId = $(this).data('id');
                console.log("Deleting faculty ID: ", facultyId);
                if (confirm('Are you sure you want to delete this faculty?')) {
                    ajaxRequest('../database/Faculty/delete_faculty.php', 'POST', { fid: facultyId }, function(response) {
                        fetchFacultyList();
                        alert(response);
                    });
                }
            });

            function fetchFacultyList() {
                ajaxRequest('../database/Faculty/show_faculty.php', 'GET', {}, function(response) {
                    $('#facultyTable tbody').html(response);
                    console.log("Data fetched: ", response);
                });
            }

            function ajaxRequest(url, method, data, successCallback) {
                $.ajax({
                    url: url,
                    type: method,
                    data: data,
                    beforeSend: function() {
                        $('button[type="submit"]').prop('disabled', true);
                    },
                    success: function(response) {
                        $('button[type="submit"]').prop('disabled', false);
                        successCallback(response);
                    },
                    error: function(xhr) {
                        console.error("AJAX error: " + xhr.responseText);
                        $('button[type="submit"]').prop('disabled', false);
                        alert("An error occurred. Please try again later.");
                    }
                });
            }
        });
    </script>
